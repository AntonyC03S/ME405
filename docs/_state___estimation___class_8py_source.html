1"""! @file discretized_estimation.py
2@brief Discrete-time state observer for the Romi robot.
3
4This module implements a Luenberger-style state observer for the Romi robot
5using discrete-time system matrices and `ulab` for lightweight linear algebra
6in MicroPython.
7
8The observer estimates the state vector
9
10@f[
11x = \\begin{bmatrix}
12\\omega_L \\\\
13\\omega_R \\\\
14s \\\\
15\\psi
16\\end{bmatrix}
17@f]
18
19where:
20
21- @f$ \\omega_L, \\omega_R @f$ are left/right wheel angular speeds
22- @f$ s @f$ is (approximately) the forward arc length / position
23- @f$ \\psi @f$ is the heading (yaw angle, in radians)

30class RomiObserver:
31 """! Discrete-time Romi state observer.
32
33 @brief Estimates robot states from motor commands and sensor feedback.
34
35 The observer uses a discrete-time linear model:
36
37 @f[
38 x_{k+1} = A_d x_k + B_d u_k + L (y_k - C x_k)
39 @f]
40
41 with measurements:
42
43 - @f$ S_L, S_R @f$: left/right encoder arc lengths [m]
44 - @f$ \\psi @f$: IMU heading [rad]
45 - @f$ \\dot{\\psi} @f$: IMU yaw rate [rad/s]
46
47 and state:
48
49 - @f$ x = [\\omega_L, \\omega_R, s, \\psi]^T @f$
50 """
51
52 def __init__(self, Ad, Bd, r: float = 0.035, l: float = 0.141, L=None):
53 """! Initialize the Romi observer.
54
55 @param Ad Discrete-time A matrix (currently unused; internal default is used).
56 @param Bd Discrete-time B matrix (currently unused; internal default is used).
57 @param r Wheel radius [m] (default: 0.035).
58 @param l Wheel track width [m] (default: 0.141).
59 @param L Observer gain matrix (4x4). If None, a default gain is used.
60
61 @note Currently, the constructor uses built-in Ad and Bd matrices tuned
62 for a specific sampling period and robot model. The @p Ad and @p Bd
63 parameters are present for future flexibility but are not applied.
64 """
65 # Discrete-time system matrices (pre-computed offline)
66 self.Ad = np.array(
67 [
68 [0.904837, 0, 0, 0],
69 [0, 0.904837, 0, 0],
70 [0.000167, 0.000167, 1, 0],
71 [-0.002362, 0.002362, 0, 1],
72 ]
73 )
74 self.Bd = np.array(
75 [
76 [0.553633, 0],
77 [0, 0.553633],
78 [0.000049, 0.000049],
79 [-0.000699, 0.000699],
80 ]
81 )
82
83 # Output matrix maps state -> measured outputs [SL, SR, psi, psidot]
84 self.C = np.array(
85 [
86 [0, 0, 1, -l / 2], # SL
87 [0, 0, 1, l / 2], # SR
88 [0, 0, 0, 1], # psi
89 [-r / l, r / l, 0, 0] # psidot
90 ]
91 )
92
93 if L is None:
94 # Starter gain (tune on-robot: larger -> faster correction, but noisier)
95 self.L = np.array(
96 [
97 [0.08, -0.08, 0.00, -0.02],
98 [-0.08, 0.08, 0.00, 0.02],
99 [0.35, 0.35, 0.00, 0.00],
100 [0.00, 0.00, 0.50, 0.70],
101 ]
102 )
103 else:
104 self.L = np.array(L)
105
106 # State vector x = [wL, wR, s, psi]^T
107 self.x = np.zeros((4, 1))
108
109 def reset(self, x0=None) -> None:
110 """! Reset the observer state.
111
112 @param x0 Optional initial state vector (4x1). If None, zeros are used.
113
114 This is useful when re-starting an experiment or re-initializing
115 the estimate after a large disturbance.
116 """
117 self.x = np.array(x0) if x0 is not None else np.zeros((4, 1))

119 def step(self, u, SL, SR, psi_deg: float, yawrate_dps: float):
120 """! Advance the observer one time step.
121
122 @brief Updates the state estimate based on input and sensor readings.
123
124 @param u Control input vector (2x1) with motor voltages [[uL], [uR]].
125 @param SL Left wheel arc length [m] from encoder.
126 @param SR Right wheel arc length [m] from encoder.
127 @param psi_deg IMU heading [deg].
128 @param yawrate_dps IMU yaw rate [deg/s].
129 @return Updated state estimate @f$ x_{k+1} @f$ as a (4x1) array.
130
131 The method:
132 - Converts IMU angles from degrees to radians
133 - Forms measurement vector y = [SL, SR, psi, psidot]^T
134 - Computes output estimate yhat = C x
135 - Applies observer correction using the gain L
136 """
137 # Convert IMU readings to radians / radians per second
138 psi = psi_deg * (pi / 180.0)
139 psidot = yawrate_dps * (pi / 180.0)
140
141 # Measurement vector
142 y = np.array([[SL], [SR], [psi], [psidot]])
143
144 x = self.x
145 yhat = np.dot(self.C, x)
146 e = y - yhat
147
148 # Observer update: x_{k+1} = A_d x_k + B_d u_k + L (y - C x_k)
149 x_next = np.dot(self.Ad, x) + np.dot(self.Bd, u) + np.dot(self.L, e)
150 self.x = x_next
151 return x_next
152
153 def state(self):
154 """! Get the current state estimate.
155
156 @return State vector x as (4x1) array:
157 [wL, wR, s, psi]^T.
158 """
159 return self.x
160
161 def outputs_hat(self):
162 """! Get estimated outputs based on current state.
163
164 @brief Computes @f$ \\hat{y} = C x @f$.
165
166 @return Estimated output vector (4x1):
167 [SL_hat, SR_hat, psi_hat, psidot_hat]^T.
168 """
169 return np.dot(self.C, self.x)
