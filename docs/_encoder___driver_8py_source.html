53 def update(self):
54 """! Update the encoder state.
55
56 @brief Reads the timer counter, handles overflow/underflow and
57 updates position, delta, and time step.
58 """
59 now_count = self._tim.counter()
60 self._delta = ticks_diff(self._prev_count, now_count)
61 self._prev_count = now_count
62
63 # Overflow: delta < -(AR+1)/2; correct by adding AR + 1
64 if self._delta < -(self._AR + 1) / 2:
65 self._delta += self._AR + 1
66
67 # Underflow: delta > (AR+1)/2; correct by subtracting AR + 1
68 elif self._delta > (self._AR + 1) / 2:
69 self._delta -= self._AR + 1
70
71 self._position += self._delta
72 now = ticks_us()
73 self._dt = ticks_diff(now, self._prev_time)
74 self._prev_time = now
