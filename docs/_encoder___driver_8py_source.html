70 def update(self):
71 """! Update the encoder state.
72
73 @brief Reads the timer counter, handles overflow/underflow and
74 updates position, delta, and time step.
75
76 This method must be called periodically in the main loop to keep
77 @ref position and @ref velocity up-to-date.
78 """
79 # Read current timer count
80 now_count = self._tim.counter()
81
82 # Compute delta in counts, accounting for wraparound
83 self._delta = ticks_diff(self._prev_count, now_count)
84 self._prev_count = now_count
85
86 # Overflow: delta < -(AR+1)/2; correct by adding AR + 1
87 if self._delta < -(self._AR + 1) / 2:
88 self._delta += self._AR + 1
89
90 # Underflow: delta > (AR+1)/2; correct by subtracting AR + 1
91 elif self._delta > (self._AR + 1) / 2:
92 self._delta -= self._AR + 1
93
94 # Integrate position
95 self._position += self._delta
96
97 # Update timing
98 now = ticks_us()
99 self._dt = ticks_diff(now, self._prev_time)
100 self._prev_time = now

113 def distance_traveled(self, radius: float = 35.0, ticks_per_rev: float = 1437.1) -> float:
114 """! Get absolute wheel distance traveled.
115
116 @brief Computes the total linear distance the wheel has traveled.
117
118 @param radius Wheel radius in millimeters (default: 35 mm).
119 @param ticks_per_rev Encoder ticks per mechanical revolution
120 (default: 1437.1).
121 @return Distance traveled in meters.
122
123 The distance is computed as:
124
125 @f[
126 d = 2 \pi r \cdot \frac{N}{\text{ticks\_per\_rev}}
127 @f]
128
129 where @f$ N @f$ is the accumulated encoder count.
130 """
131 rev = self._position / float(ticks_per_rev)
132 dist_m = 2 * math.pi * (radius / 1000.0) * rev
133 return dist_m
