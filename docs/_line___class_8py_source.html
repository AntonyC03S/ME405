179 def update(self) -> float:
180 """! Compute line position using centroid.
181
182 @brief Uses normalized sensor readings to compute a weighted average
183 of sensor positions, yielding an estimated line position.
184
185 @return Centroid position as a float in the same index space as
186 @ref positions (e.g. between 1 and 15 for an 8-sensor setup).
187 """
188 values = self.readings()
189
190 centroid_max = 15
191 centroid_min = 1
192 centroid_middle = 8
193
194 numerator = 0.0
195 denominator = 0.0
196
197 for i in range(len(values)):
198 numerator += values[i] * self.positions[i]
199 denominator += values[i]
200
201 # If all sensors read zero, assume line is centered
202 if denominator == 0:
203 return centroid_middle
204
205 centroid = numerator / denominator
206
207 if centroid > centroid_max:
208 return centroid_max
209 elif centroid < centroid_min:
210 return centroid_min
211
212 return centroid

214 def readings(self):
215 """! Get normalized sensor readings.
216
217 @brief Returns a list of calibrated, thresholded sensor values.
218
219 Each value is:
220 - Calibrated between black and white references
221 - Flipped so higher value means "darker" (more line)
222 - Thresholded such that very small values are set to 0
223
224 @return List of floats in approximately [0, 1].
225 """
226 return [
227 self.color_def(abs(self.calibrate(adc.read(), idx) - 1))
228 for idx, adc in enumerate(self._sensors)

232 def calibrate(self, value: int, idx: int) -> float:
233 """! Map raw ADC value into a normalized range [0, 1].
234
235 @brief Uses stored black/white calibration values to normalize a reading.
236
237 @param value Raw ADC reading for a single sensor.
238 @param idx Index of the sensor in the array.
239 @return Normalized value between 0 (white) and 1 (black).
240 """
241 # Lazy-load calibration from file if needed
242 if self._white_cal is None or self._black_cal is None:
243 filelist = listdir()
244 if "calibration.txt" in filelist:
245 # Calibration data is present
246 print("Found calibration data, skipping calibration")
247 with open("calibration.txt", "r") as file:
248 lines = file.readlines()
249 self._white_cal = [int(x) for x in eval(lines[0])]
250 self._black_cal = [int(x) for x in eval(lines[1])]
251 else:
252 # Calibration data is not present; do interactive calibration
253 input("Accept Black?")
254 self.cali_black()
255 input("Accept White?")
256 self.cali_white()
257
258 with open("calibration.txt", "w") as file:
259 file.write(", ".join(str(item) for item in self._white_cal))
260 file.write("\n")
261 file.write(", ".join(str(item) for item in self._black_cal))
262
263 return (self._black_cal[idx] - value) / (
264 self._black_cal[idx] - self._white_cal[idx]
265 )
