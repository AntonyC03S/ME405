12class IMU:
13 """! BNO055 IMU driver over I2C.
14
15 @brief Wraps common BNO055 functions such as mode switching, calibration,

20 # Register addresses
21 OPR_MODE_REG = 0x3D
22 PWR_MODE_REG = 0x3E
23 SYS_TRIGGER_REG = 0x3F
24 CHIP_ID_REG = 0x00
25 PAGE_ID_REG = 0x07
26 ACC_DATA_X_LSB = 0x08
27 MAG_DATA_X_LSB = 0x0E
28 GYRO_DATA_X_LSB = 0x14
29 EULER_H_LSB = 0x1A
30 CAL_DATA_LSB = 0x55
31
32 # Operation modes
33 CONFIG_MODE = 0x00
34 NDOF_MODE = 0x0C
35 IMU_MODE = 0x08

37 def __init__(self, i2c: I2C, addr: int = 0x28):
38 """! Initialize an IMU instance.
39
40 @param i2c Initialized I2C object.
41 @param addr 7-bit I2C address of the BNO055 (default: 0x28).
42 """
43 self.i2c = i2c
44 self.devad = addr
45 self._set_mode("CONFIG")
46
47 def _set_mode(self, mode: str) -> None:
48 """! Internal helper to switch IMU operation mode.
49
50 @param mode String specifying mode: `"CONFIG"`, `"NDOF"`, or `"IMU"`.
51
52 """
53 if mode == "NDOF":
54 mode_val = self.NDOF_MODE
55 elif mode == "CONFIG":
56 mode_val = self.CONFIG_MODE
57 elif mode == "IMU":
58 mode_val = self.IMU_MODE
59 else:
60 mode_val = self.CONFIG_MODE
61
62 self.i2c.mem_write(bytes([0x00]), self.devad, self.PAGE_ID_REG)
63 time.sleep_ms(10)
64
65 self.i2c.mem_write(bytes([mode_val]), self.devad, self.OPR_MODE_REG)
66 time.sleep_ms(25)

68 def cali_status(self):
69 """! Get IMU calibration status.
70
71 @brief Returns the calibration state of system, gyro, accelerometer,
72 and magnetometer.
73
74 @return Tuple `(sys, gyro, acc, mag)` where each value is from 0-3.
75 A value of 3 indicates fully calibrated.
76 """
77 cal_stat = self.i2c.mem_read(1, self.devad, 0x35)[0]
78 sys = (cal_stat >> 6) & 0x03
79 gyro = (cal_stat >> 4) & 0x03
80 acc = (cal_stat >> 2) & 0x03
81 mag = cal_stat & 0x03
82 return (sys, gyro, acc, mag)

84 def read_euler(self):
85 """! Read Euler angles from the IMU.
86
87 @brief Returns heading, roll, and pitch in degrees.
88
89 @return Tuple `(heading, roll, pitch)` in degrees.
90
91 """
92 raw = self.i2c.mem_read(6, self.devad, self.EULER_H_LSB)
93 heading = struct.unpack('<h', raw[0:2])[0] / 16.0
94 roll = struct.unpack('<h', raw[2:4])[0] / 16.0
95 pitch = struct.unpack('<h', raw[4:6])[0] / 16.0
96 return heading, roll, pitch

98 def read_heading(self) -> float:
99 """! Read only the heading (yaw) angle.
100
101 @return Heading in degrees.
102 """
103 raw = self.i2c.mem_read(2, self.devad, self.EULER_H_LSB)
104 heading = struct.unpack('<h', raw[0:2])[0] / 16.0
105 return heading

107 def get_cal_data(self, filename: str = "cal_data.bin"):
108 """! Read and store calibration data to a file.
109
110 @brief Switches to CONFIG mode, reads calibration data, and writes
111 it to a binary file for later reuse.
112
113 @param filename Name of the file where calibration data is saved.
114 Default is `"cal_data.bin"`.
115
116 @return The raw calibration data as a bytes object.
117 """
118 # Save current mode
119 prev = self.i2c.mem_read(1, self.devad, self.OPR_MODE_REG)[0]
120
121 # Switch to config mode and page 0
122 self.i2c.mem_write(bytes([self.CONFIG_MODE]), self.devad, self.OPR_MODE_REG)
123 time.sleep_ms(25)
124 self.i2c.mem_write(bytes([0x00]), self.devad, self.PAGE_ID_REG)
125 time.sleep_ms(5)
126
127 # Read 22 bytes of calibration data from 0x55
128 data = self.i2c.mem_read(22, self.devad, self.CAL_DATA_LSB)
129
130 # Store to file
131 with open(filename, "wb") as f:
132 f.write(data)
133
134 # Restore previous mode
135 self.i2c.mem_write(bytes([prev]), self.devad, self.OPR_MODE_REG)
136 time.sleep_ms(25)
137
138 return data

140 def push_cal_data(self, filename: str = "cal_data.bin") -> None:
141 """! Push stored calibration data into the IMU.
142
143 @brief Reads calibration data from a file and writes it to the IMU.
144
145 @param filename File containing previously saved calibration data
146 (default: `"cal_data.bin"`).
147
148 This can be used at startup to restore a known good calibration,
149 avoiding re-calibration on every power cycle.
150 """
151 with open(filename, "rb") as f:
152 file_content_bytes = f.read()
153
154 cal_data = bytearray(file_content_bytes)
155 self.i2c.mem_write(cal_data, self.devad, self.CAL_DATA_LSB)

157 def read_gyro(self):
158 """! Read gyroscope rates for all three axes.
159
160 @brief Reads raw gyro data from the IMU and converts it to deg/s.
161
162 @return Tuple `(gx, gy, gz)` in degrees per second.
163 """
164 raw = self.i2c.mem_read(6, self.devad, self.GYRO_DATA_X_LSB)
165 gx = struct.unpack('<h', raw[0:2])[0] / 16.0
166 gy = struct.unpack('<h', raw[2:4])[0] / 16.0
167 gz = struct.unpack('<h', raw[4:6])[0] / 16.0
168 return gx, gy, gz

171 """! Read yaw rate from the gyroscope.
172
173 @brief Returns the Z-axis gyro rate.
