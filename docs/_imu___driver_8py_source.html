18class IMU:
19 """! BNO055 IMU driver over I2C.
20
21 @brief Wraps common BNO055 functions such as mode switching, calibration,

26 # Register addresses
27 OPR_MODE_REG = 0x3D
28 PWR_MODE_REG = 0x3E
29 SYS_TRIGGER_REG = 0x3F
30 CHIP_ID_REG = 0x00
31 PAGE_ID_REG = 0x07
32 ACC_DATA_X_LSB = 0x08
33 MAG_DATA_X_LSB = 0x0E
34 GYRO_DATA_X_LSB = 0x14
35 EULER_H_LSB = 0x1A
36 CAL_DATA_LSB = 0x55
37
38 # Operation modes
39 CONFIG_MODE = 0x00
40 NDOF_MODE = 0x0C
41 IMU_MODE = 0x08

43 def __init__(self, i2c: I2C, addr: int = 0x28):
44 """! Initialize an IMU instance.
45
46 @param i2c Initialized I2C object.
47 @param addr 7-bit I2C address of the BNO055 (default: 0x28).
48
49 The constructor sets the IMU into CONFIG mode initially.
50 """
51 self.i2c = i2c
52 self.devad = addr
53 self._set_mode("CONFIG")
54
55 def _set_mode(self, mode: str) -> None:
56 """! Internal helper to switch IMU operation mode.
57
58 @param mode String specifying mode: `"CONFIG"`, `"NDOF"`, or `"IMU"`.
59
60 This writes to the operation mode register and includes required
61 delays as per the BNO055 datasheet.
62 """
63 if mode == "NDOF":
64 mode_val = self.NDOF_MODE
65 elif mode == "CONFIG":
66 mode_val = self.CONFIG_MODE
67 elif mode == "IMU":
68 mode_val = self.IMU_MODE
69 else:
70 # Default to CONFIG if an unknown mode string is given
71 mode_val = self.CONFIG_MODE
72
73 # Always use page 0 for these registers
74 self.i2c.mem_write(bytes([0x00]), self.devad, self.PAGE_ID_REG)
75 time.sleep_ms(10)
76
77 self.i2c.mem_write(bytes([mode_val]), self.devad, self.OPR_MODE_REG)
78 time.sleep_ms(25)

80 def cali_status(self):
81 """! Get IMU calibration status.
82
83 @brief Returns the calibration state of system, gyro, accelerometer,
84 and magnetometer.
85
86 @return Tuple `(sys, gyro, acc, mag)` where each value is from 0–3.
87 A value of 3 indicates fully calibrated.
88 """
89 cal_stat = self.i2c.mem_read(1, self.devad, 0x35)[0]
90 sys = (cal_stat >> 6) & 0x03
91 gyro = (cal_stat >> 4) & 0x03
92 acc = (cal_stat >> 2) & 0x03
93 mag = cal_stat & 0x03
94 return (sys, gyro, acc, mag)
95
96 def read_euler(self):
97 """! Read Euler angles from the IMU.
98
99 @brief Returns heading, roll, and pitch in degrees.
100
101 @return Tuple `(heading, roll, pitch)` in degrees.
102
103 These angles are provided directly by the BNO055 in fusion mode
104 (e.g. NDOF).
105 """
106 raw = self.i2c.mem_read(6, self.devad, self.EULER_H_LSB)
107 heading = struct.unpack('<h', raw[0:2])[0] / 16.0
108 roll = struct.unpack('<h', raw[2:4])[0] / 16.0
109 pitch = struct.unpack('<h', raw[4:6])[0] / 16.0
110 return heading, roll, pitch
111
112 def read_heading(self) -> float:
113 """! Read only the heading (yaw) angle.
114
115 @return Heading in degrees.
116 """
117 raw = self.i2c.mem_read(2, self.devad, self.EULER_H_LSB)
118 heading = struct.unpack('<h', raw[0:2])[0] / 16.0
119 return heading

121 def get_cal_data(self, filename: str = "cal_data.bin"):
122 """! Read and store calibration data to a file.
123
124 @brief Switches to CONFIG mode, reads calibration data, and writes
125 it to a binary file for later reuse.
126
127 @param filename Name of the file where calibration data is saved.
128 Default is `"cal_data.bin"`.
129
130 @return The raw calibration data as a bytes object.
131 """
132 # Save current mode
133 prev = self.i2c.mem_read(1, self.devad, self.OPR_MODE_REG)[0]
134
135 # Switch to config mode and page 0
136 self.i2c.mem_write(bytes([self.CONFIG_MODE]), self.devad, self.OPR_MODE_REG)
137 time.sleep_ms(25)
138 self.i2c.mem_write(bytes([0x00]), self.devad, self.PAGE_ID_REG)
139 time.sleep_ms(5)
140
141 # Read 22 bytes of calibration data from 0x55
142 data = self.i2c.mem_read(22, self.devad, self.CAL_DATA_LSB)
143
144 # Store to file
145 with open(filename, "wb") as f:
146 f.write(data)
147
148 # Restore previous mode
149 self.i2c.mem_write(bytes([prev]), self.devad, self.OPR_MODE_REG)
150 time.sleep_ms(25)
151
152 return data

154 def push_cal_data(self, filename: str = "cal_data.bin") -> None:
155 """! Push stored calibration data into the IMU.
156
157 @brief Reads calibration data from a file and writes it to the IMU.
158
159 @param filename File containing previously saved calibration data
160 (default: `"cal_data.bin"`).
161
162 This can be used at startup to restore a known good calibration,
163 avoiding re-calibration on every power cycle.
164 """
165 with open(filename, "rb") as f:
166 file_content_bytes = f.read()
167
168 cal_data = bytearray(file_content_bytes)
169 self.i2c.mem_write(cal_data, self.devad, self.CAL_DATA_LSB)

171 def read_gyro(self):
172 """! Read gyroscope rates for all three axes.
173
174 @brief Reads raw gyro data from the IMU and converts it to deg/s.
175
176 @return Tuple `(gx, gy, gz)` in degrees per second.
177 """
178 raw = self.i2c.mem_read(6, self.devad, self.GYRO_DATA_X_LSB)
179 gx = struct.unpack('<h', raw[0:2])[0] / 16.0
180 gy = struct.unpack('<h', raw[2:4])[0] / 16.0
181 gz = struct.unpack('<h', raw[4:6])[0] / 16.0
182 return gx, gy, gz

184 def read_yaw_rate(self) -> float:
185 """! Read yaw rate from the gyroscope.
186
187 @brief Returns the Z-axis gyro rate.
188
189 @return Yaw rate (gz) in degrees per second.
190 """
191 GYRO_Z_LSB = 0x18
192 raw = self.i2c.mem_read(2, self.devad, GYRO_Z_LSB)
193 yaw_rate = struct.unpack('<h', raw)[0] / 16.0
194 return yaw_rate
